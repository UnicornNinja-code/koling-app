### 2. Komponen Item Navigasi Rekursif: `src/components/layout/sidebar/SidebarNavItem.svelte`

Menggunakan `$state` untuk status buka/tutup sub-menu, `$derived` untuk mengecek *active state*, dan pemanggilan diri sendiri secara rekursif (`svelte:self`) untuk menangani kedalaman anak item tak terbatas.

```svelte
<script lang="ts">
  import { ChevronRight } from "lucide-svelte";
  import type { NavItemData } from "./types";

  let {
    item,
    activeId,
    onSelect,
    level = 0,
  }: {
    item: NavItemData;
    activeId: string;
    onSelect: (id: string) => void;
    level?: number;
  } = $props();

  let isOpen = $state(false);
  let isActive = $derived(activeId === item.id);
  let hasChildren = $derived(Boolean(item.children && item.children.length > 0));

  function handleClick() {
    if (hasChildren) {
      isOpen = !isOpen;
    } else {
      onSelect(item.id);
    }
  }
</script>

<div class="flex flex-col w-full">
  <button
    type="button"
    class={`group flex w-full items-center justify-between px-2.5 py-[7px] rounded-[6px] cursor-pointer transition-all duration-200 select-none text-left
      ${
        isActive
          ? "bg-black/5 dark:bg-white/10 text-foreground font-medium"
          : "text-muted-foreground hover:bg-black/5 dark:hover:bg-white/5 hover:text-foreground/90"
      }
    `}
    style="padding-left: {level * 12 + 10}px;"
    onclick={handleClick}
  >
    <div class="flex items-center gap-2.5 min-w-0">
      {#snippet renderIcon()}
        {@const Icon = item.icon}
        <Icon
          class={`size-4 shrink-0 transition-colors ${
            isActive
              ? "text-foreground"
              : "text-muted-foreground/70 group-hover:text-foreground/70"
          }`}
          strokeWidth={1.5}
        />
      {/snippet}
      {@render renderIcon()}
      <span class="text-[13px] tracking-wide truncate">
        {item.title}
      </span>
    </div>

    <div class="flex items-center gap-2 shrink-0">
      {#if item.shortcut}
        <kbd
          class="hidden group-hover:inline-flex items-center justify-center h-5 px-1.5 text-[10px] font-medium font-mono text-muted-foreground/60 bg-background/50 border border-border/50 rounded-[4px] shadow-xs"
        >
          {item.shortcut}
        </kbd>
      {/if}

      {#if item.badge !== undefined}
        <span
          class="flex items-center justify-center min-w-[20px] h-5 px-1.5 text-[10px] font-medium rounded-full bg-primary/10 text-primary"
        >
          {item.badge}
        </span>
      {/if}

      {#if hasChildren}
        <ChevronRight
          class="size-3.5 text-muted-foreground/50 transition-transform duration-200 {isOpen ? 'rotate-90' : ''}"
          strokeWidth={2}
        />
      {/if}
    </div>
  </button>

  {#if hasChildren && item.children}
    <div
      class="grid transition-[grid-template-rows,opacity] duration-300 ease-in-out {isOpen ? 'grid-rows-[1fr] opacity-100' : 'grid-rows-[0fr] opacity-0'}"
    >
      <div class="overflow-hidden min-h-0 relative flex flex-col gap-0.5 mt-0.5">
        <div
          class="absolute top-0 bottom-0 border-l border-black/5 dark:border-white/5"
          style="left: {level * 12 + 17.5}px;"
        ></div>
        {#each item.children as child (child.id)}
          <svelte:self
            item={child}
            {activeId}
            {onSelect}
            level={level + 1}
          />
        {/each}
      </div>
    </div>
  {/if}
</div>

```

```svelte
<script lang="ts">
  import { ChevronDown } from "lucide-svelte";

  let {
    selected = $bindable("Acme Corp"),
    onSelect,
  }: {
    selected?: string;
    onSelect?: (ws: string) => void;

  let isOpen = $state(false);

    selected = ws;
```

### 5. Komponen Demo Interaktif Lengkap: `src/components/demo/DashboardSidebarDemo.svelte`

Mengintegrasikan tombol collapse sidebar, breadcrumb workspace aktif, dan modal pencarian (*Command Search Palette*).

```svelte
<script lang="ts">
  import DashboardSidebar from "$components/layout/sidebar/DashboardSidebar.svelte";
  import { PanelLeftClose, PanelLeftOpen, Search, Command, X } from "lucide-svelte";

  let isOpen = $state(true);
  let activeId = $state("home");
  let activeWorkspace = $state("Acme Corp");
  let isSearchOpen = $state(false);

  function handleSelect(id: string) {
    if (id === "search") {
      isSearchOpen = true;
      return;
    }
    activeId = id;
  }

  // Keyboard shortcut cmd+k / ctrl+k untuk membuka pencarian
  function handleKeyDown(e: KeyboardEvent) {
    if ((e.metaKey || e.ctrlKey) && e.key === "k") {
      e.preventDefault();
      isSearchOpen = !isSearchOpen;
    }
    if (e.key === "Escape") {
      isSearchOpen = false;
    }
  }
</script>
```

### 6. Integrasi dengan Role-Based Sidebar MOVA (FE PART 00)

Struktur data `navGroups` di atas dapat langsung disesuaikan secara dinamis berdasarkan peran yang aktif (`user.role` dari `auth.svelte.ts`):
